var SCALE_10M = 10;

var SCALE_20M = 20;

var DRIVE_FOLDER = 'Tehran22_Sentinel2';

var MIN_COVERAGE = 99;

var BUFFER_DISTANCE = 40;

  'B3',
  'B4',
  'B8'

  'B11',
  'B8',
  'B4',

// ==========================================================
// 4. TOTAL ROI PIXEL COUNT
// Used for the original ROI coverage calculation
// ==========================================================

var totalPixelCount = ee.Number(

  ee.Image.constant(1)
    .rename('total')

    .reduceRegion({

      reducer: ee.Reducer.sum(),

      geometry: roiGeometry,

      scale: SCALE_20M,

      crs: CRS,

      maxPixels: 1e10,

      tileScale: 4

    })

    .get('total')

);

print('Total ROI pixels:', totalPixelCount);


// ==========================================================
// 5. CALCULATE IMAGE QUALITY
// Original selection criteria preserved
// ==========================================================

function addImageQuality(img) {

  img = ee.Image(img);

  var scl = img.select('SCL');


  // Classified pixels: SCL 1 to 11

  var validMask = scl.gte(1)
    .and(scl.lte(11));


  // Clear pixels: vegetation, bare soil, water

  var clearMask = scl.eq(4)
    .or(scl.eq(5))
    .or(scl.eq(6));


  var statsImage = ee.Image.cat([

    ee.Image.constant(1)
      .rename('valid')
      .updateMask(validMask),

    ee.Image.constant(1)
      .rename('clear')
      .updateMask(clearMask)

  ]);


  var stats = statsImage.reduceRegion({

    reducer: ee.Reducer.sum(),

    geometry: roiGeometry,

    scale: SCALE_20M,

    crs: CRS,

    maxPixels: 1e10,

    tileScale: 4

  });


  var validCount = ee.Number(
    stats.get('valid')
  );

  var clearCount = ee.Number(
    stats.get('clear')
  );


  var coveragePercentage = validCount
    .divide(totalPixelCount)
    .multiply(100);


  var clearPercentage = clearCount
    .divide(totalPixelCount)
    .multiply(100);


  return img.set({

    'ROI_COVERAGE_PCT': coveragePercentage,

    'ROI_CLEAR_PCT': clearPercentage

  });

}


// ==========================================================
// 6. MOSAIC GRANULES OF THE SAME ACQUISITION
// Restore native projections after mosaic()
// ==========================================================

function createAcquisitionMosaics(collection) {

  var identifiers = ee.List(

    collection.aggregate_array(
      'DATATAKE_IDENTIFIER'
    )

  ).distinct();


  var mosaics = ee.ImageCollection.fromImages(

    identifiers.map(function(identifier) {

      var group = collection.filter(

        ee.Filter.eq(
          'DATATAKE_IDENTIFIER',
          identifier
        )

      );


      var first = ee.Image(
        group.first()
      );

      var mosaic = ee.Image(
        group.mosaic()
      );


      // Native projections

      var projection10m = first
        .select('B2')
        .projection();

      var projection20m = first
        .select('B11')
        .projection();


      // Restore 10m bands

      var bands10m = mosaic

        .select([
          'B8',
          'B4',
          'B3',
          'B2'
        ])

        .setDefaultProjection(
          projection10m
        );


      // Restore 20m bands

      var bands20m = mosaic

        .select([
          'B11',
          'B12'
        ])

        .setDefaultProjection(
          projection20m
        );


      // Restore SCL projection

      var sclBand = mosaic

        .select('SCL')

        .setDefaultProjection(
          projection20m
        );


      // Combine all bands

      var fixedMosaic = bands10m

        .addBands(bands20m)

        .addBands(sclBand);


      return ee.Image(fixedMosaic)

        .copyProperties(first, [

          'system:time_start',

          'DATATAKE_IDENTIFIER',

          'CLOUDY_PIXEL_PERCENTAGE'

        ])

        .set(
          'DATATAKE_IDENTIFIER',
          identifier
        );

    })

  );


  return mosaics;

}


// ==========================================================
// 7. SELECT THE BEST JULY ACQUISITION
// Selection is performed on the ORIGINAL ROI
// ==========================================================

function getBestJulyImage(year) {

  var startDate = ee.Date.fromYMD(
    year,
    JULY,
    1
  );

  var endDate = startDate.advance(
    1,
    'month'
  );


  var collection = S2

    .filterBounds(roiGeometry)

    .filterDate(
      startDate,
      endDate
    )

    .filter(
      ee.Filter.notNull([
        'DATATAKE_IDENTIFIER'
      ])
    );


  print(
    'Year ' + year + ' - Original granules:',
    collection.size()
  );


  var mosaics = createAcquisitionMosaics(
    collection
  );


  var ranked = mosaics

    .map(addImageQuality)

    .filter(
      ee.Filter.gte(
        'ROI_COVERAGE_PCT',
        MIN_COVERAGE
      )
    )

    .sort(
      'ROI_CLEAR_PCT',
      false
    );


  return {

    collection: collection,

    ranked: ranked

  };

}


// ==========================================================
// 8. CREATE 10m IMAGE
// B2, B3, B4, B8
// Reflectance scaled to 0-1
// ==========================================================

function make10mImage(sourceGranules) {

  var first = ee.Image(
    sourceGranules.first()
  );


  var projection10m = first
    .select('B2')
    .projection();


  var image10m = sourceGranules

    .select(BANDS_10M)

    .mosaic()

    .setDefaultProjection(
      projection10m
    )

    .divide(10000)

    .toFloat()

    .select(BANDS_10M)

    .clip(exportGeometry);


  return image10m;

}


// ==========================================================
// 9. CREATE 20m IMAGE
// B11, B8, B4, B3, B2, B12, SCL
//
// 10m bands aggregated to 20m using mean
// 20m bands retain native resolution
// SCL remains categorical but cast to Float32 for export
// Clear mask: SCL 4, 5, 6
// ==========================================================

function make20mImage(sourceGranules) {

  var first = ee.Image(
    sourceGranules.first()
  );


  var projection10m = first
    .select('B2')
    .projection();

  var projection20m = first
    .select('B11')
    .projection();


  // --------------------------------------------------------
  // Native 10m bands
  // --------------------------------------------------------

  var image10m = sourceGranules

    .select([
      'B8',
      'B4',
      'B3',
      'B2'
    ])

    .mosaic()

    .setDefaultProjection(
      projection10m
    );


  // --------------------------------------------------------
  // Native SCL band
  // --------------------------------------------------------

  var scl = sourceGranules

    .select('SCL')

    .mosaic()

    .setDefaultProjection(
      projection20m
    );


  // Clear pixel mask

  var clearMask = scl.eq(4)
    .or(scl.eq(5))
    .or(scl.eq(6));


  // --------------------------------------------------------
  // Aggregate 10m bands to 20m
  // --------------------------------------------------------

  var bands10mTo20m = image10m

    .updateMask(clearMask)

    .divide(10000)

    .reduceResolution({

      reducer: ee.Reducer.mean(),

      maxPixels: 16

    })

    .reproject({

      crs: CRS,

      scale: SCALE_20M

    });


  // --------------------------------------------------------
  // Native 20m spectral bands
  // --------------------------------------------------------

  var bands20m = sourceGranules

    .select([
      'B11',
      'B12'
    ])

    .mosaic()

    .setDefaultProjection(
      projection20m
    )

    .updateMask(clearMask)

    .divide(10000)

    .reproject({

      crs: CRS,

      scale: SCALE_20M

    });


  // --------------------------------------------------------
  // SCL categorical band
  // Cast to Float32 so all exported bands share
  // the same data type (fixes inconsistent types error)
  // --------------------------------------------------------

  var sclOutput = scl

    .reproject({

      crs: CRS,

      scale: SCALE_20M

    })

    .updateMask(clearMask)

    .toFloat();


  // --------------------------------------------------------
  // Combine bands in requested order
  // --------------------------------------------------------

  var finalImage = bands20m

    .select('B11')

    .addBands(
      bands10mTo20m.select([
        'B8',
        'B4',
        'B3',
        'B2'
      ])
    )

    .addBands(
      bands20m.select('B12')
    )

    .addBands(sclOutput)

    .select(BANDS_20M)

    .clip(exportGeometry);


  // Safety cast: ensure every band is Float32

  return finalImage.toFloat();

}

// ==========================================================
// 11. EXPORT METADATA CSV
// One combined metadata table for both years
// ==========================================================

function getYearMetadata(year) {

  var result = getBestJulyImage(year);

  var ranked = result.ranked;

  var hasImage = ranked.size().gt(0);


  var fallback = ee.Image.constant(0)
    .rename('dummy')
    .set('DATATAKE_IDENTIFIER', 'NO_IMAGE');


  var bestImage = ee.Image(
    ee.Algorithms.If(
      hasImage,
      ranked.first(),
      fallback
    )
  );


  var acquisitionId = bestImage.get(
    'DATATAKE_IDENTIFIER'
  );


  var sourceGranules = S2

    .filterBounds(exportGeometry)

    .filterDate(

      ee.Date.fromYMD(year, JULY, 1),

      ee.Date.fromYMD(year, JULY, 1)
        .advance(1, 'month')

    )

    .filter(
      ee.Filter.eq(
        'DATATAKE_IDENTIFIER',
        acquisitionId
      )
    );


  var rows = sourceGranules.map(
    function(image) {

      image = ee.Image(image);

      return ee.Feature(null, {

        'YEAR': year,

        'MONTH': 'July',

        'COLLECTION':
          'COPERNICUS/S2_SR_HARMONIZED',

        'DATATAKE_IDENTIFIER':
          image.get('DATATAKE_IDENTIFIER'),

        'SYSTEM_INDEX':
          image.get('system:index'),

        'PRODUCT_ID':
          image.get('PRODUCT_ID'),

        'GRANULE_ACQUISITION_DATE_UTC':
          ee.Date(
            image.get('system:time_start')
          ).format('YYYY-MM-dd HH:mm:ss'),

        'SELECTED_ACQUISITION_DATE_UTC':
          ee.Date(
            bestImage.get('system:time_start')
          ).format('YYYY-MM-dd HH:mm:ss'),

        'ROI_COVERAGE_PERCENT':
          bestImage.get('ROI_COVERAGE_PCT'),

        'ROI_CLEAR_PERCENT':
          bestImage.get('ROI_CLEAR_PCT'),

        'BUFFER_METERS': 40,

        'CRS': CRS,

        'EXPORT_10M_BANDS':
          'B2, B3, B4, B8',

        'EXPORT_10M_SCALE': 10,

        'EXPORT_20M_BANDS':
          'B11, B8, B4, B3, B2, B12, SCL',

        'EXPORT_20M_SCALE': 20

      });

    }
  );


  return ee.FeatureCollection(
    ee.Algorithms.If(
      hasImage,
      rows,
      ee.FeatureCollection([])
    )
  );

}


var metadata2017 = getYearMetadata(2017);

var metadata2026 = getYearMetadata(2026);


var finalMetadata = metadata2017.merge(
  metadata2026
);


print(
  'FINAL METADATA TABLE:',

);

print(
  'TOTAL METADATA ROWS:',
  finalMetadata.size()
);


Export.table.toDrive({

  collection: finalMetadata,

  description:
    'Tehran22_July_2017_2026_Metadata',

  folder: DRIVE_FOLDER,

  fileNamePrefix:
    'Tehran22_July_2017_2026_Metadata',

  fileFormat: 'CSV'

});
